# ggplot2 1: average sleep duration by occupation using bar plot
plot1 <- ggplot(sleep_data, aes(x = Occupation, y = `Sleep Duration`)) +
  geom_bar(stat = "summary", fun = "mean") +
  theme(axis.text.x = element_text(angle = 45, hjust = 1)) +
  labs(title = "Average Sleep Duration by Occupation",
       x = "Occupation", y = "Average Sleep Duration (hours)")

print(plot1)

## This bar plot is more effective because it shows the average sleep duration across all occupation. It is less effective because they are many occupations so it will be hard to see the difference.
# ggplot2 2: sleep quality vs. physical activity level using scatter plot
plot2 <- ggplot(sleep_data, aes(x = `Physical Activity Level`, y = `Quality of Sleep`, 
                                color = `BMI Category`)) + 
  geom_point() +
  geom_smooth(method = "lm", se = FALSE) +
  labs(title = "Sleepy Quality vs. Physical Activity Level",
       x = "Sleep Quality", y = "Physical Activity Level")
print(plot2)
## `geom_smooth()` using formula = 'y ~ x'

## This scatter plot is more effective because it shows the physical activity level and the sleep quality also with the bmi category. It is less effective because a large number of data points can overlap.
# ggplot2 3: stress levels by sleep disorder using box plot
plot3 <- ggplot(sleep_data, aes(x = `Sleep Disorder`, y = `Stress Level`, 
                                fill = `Sleep Disorder`)) +
  geom_boxplot() +
  labs(title = "Stress Levels by Sleep Disorder",
       x = "Sleep Disorder", y = "Stress Level") +
  theme(legend.position = "none")

print(plot3)

## This box plot is more effective because it shows the distributions of stress level for each disorder. It can be misleading with less data points and thus can lack the ability to show detail.
